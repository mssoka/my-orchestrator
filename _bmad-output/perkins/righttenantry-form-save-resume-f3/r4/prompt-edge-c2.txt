+//// F3 save-and-resume: public draft endpoints.
+////
+////   PUT/POST /apply/:code/draft            — debounced field-value upsert
+////   POST     /apply/:code/continue-link    — "email me a link to continue"
+////   GET      /apply/:code/resume/:token    — restore the draft into the form
+////   GET/POST /apply/:code/draft-erasure/:token — self-service draft delete
+////
+//// Security / abuse model (mirrors the erasure endpoint):
+////   - Drafts are keyed (vacancy, lower(email)); the 288-bit continue token
+////     IS the auth for resume/erasure — no session, no CSRF (should_skip).
+////   - The save endpoint writes but never reads back: worst case an attacker
+////     stores field values against an email they don't own, visible only to
+////     whoever holds that inbox (the continue link goes to the email, not
+////     the requester).
+////   - The continue-link endpoint always answers 200; the 10-minute
+////     per-draft cooldown (claimed atomically in SQL) bounds email spam.
+////   - Field values are filtered server-side: no `_`-prefixed internals
+////     (honeypot/timing), no `gdpr_consent` (consent is captured at submit
+////     only), caps on key count/length and value length, and never any
+////     file data (the JSON payload couldn't carry it anyway).
+

+/// Server-side field-payload caps. The serializer sends ~40 fields for a
+/// full household form; 120 keys / 4000-char values leave generous room
+/// (personal_statement truncates at 2000 on submit) without allowing
+/// megabyte values against the 64 KB body gate.
+const max_field_keys = 120
+
+const max_key_length = 64
+
+const max_value_length = 4000
+
+pub type DraftPayload {
+  DraftPayload(
+    email: String,
+    fields: Dict(String, String),
+    /// Present only on the resume render (the page carries it so the first
+    /// post-resume save can stamp `resumed_at` — re-engagement, not the
+    /// prefetch-able GET, is what counts as a resume).
+    resume_token: String,
+  )
+}

+fn draft_payload_decoder() -> decode.Decoder(DraftPayload) {
+  use email <- decode.field("email", decode.string)
+  use fields <- decode.optional_field(
+    "fields",
+    dict.new(),
+    decode.dict(decode.string, decode.string),
+  )
+  use resume_token <- decode.optional_field("resume_token", "", decode.string)
+  decode.success(DraftPayload(email:, fields:, resume_token:))
+}

+/// Upload-slot field names (documents step). The JS serializer already
+/// skips type=file controls; the server filter drops these names too, so a
+/// hand-crafted JSON payload can't park anything under a document slot's
+/// identity — drafts hold form VALUES, and slots are documents.
+pub fn file_slot_names() -> List(String) {
+  [
+    "proof_of_income_primary",
+    "landlord_ref_primary",
+    "character_ref_primary",
+    "employer_ref_primary",
+    "additional_docs",
+  ]
+}
+
+const coapplicant_slot_prefixes = [
+  "proof_of_income_coapplicant_", "employer_ref_coapplicant_",
+]
+
+fn is_file_slot_key(key: String) -> Bool {
+  list.contains(file_slot_names(), key)
+  || list.any(coapplicant_slot_prefixes, fn(prefix) {
+    string.starts_with(key, prefix)
+  })
+}

+/// Filter a client-sent field map down to what a draft may hold. Pure and
+/// pub for unit tests. Drops `_`-prefixed internals (`_website` honeypot,
+/// `_form_loaded_at` timing), the consent checkbox, and every upload-slot
+/// key; enforces the caps.
+pub fn filter_fields(fields: Dict(String, String)) -> List(#(String, String)) {
+  fields
+  |> dict.to_list
+  |> list.filter(fn(pair) {
+    let #(key, _) = pair
+    !string.starts_with(key, "_")
+    && key != "gdpr_consent"
+    && !is_file_slot_key(key)
+    && string.length(key) > 0
+    && string.length(key) <= max_key_length
+  })
+  |> list.take(max_field_keys)
+  |> list.map(fn(pair) {
+    let #(key, value) = pair
+    #(key, string.slice(value, 0, max_value_length))
+  })
+}
+
+fn fields_to_json(fields: List(#(String, String))) -> json.Json {
+  json.object(list.map(fields, fn(pair) { #(pair.0, json.string(pair.1)) }))
+}

+fn json_ok(ok: Bool) -> String {
+  json.object([#("ok", json.bool(ok))]) |> json.to_string

+/// only when the vacancy exists AND is accepting applications (drafts must

+fn lookup_accepting_vacancy(

+  code: String,

+          ))
+        True ->
+          case uuid.from_string(vacancy.id) {

+/// PUT/POST /apply/:code/draft — upsert the applicant's field values.
+/// POST is accepted alongside PUT because navigator.sendBeacon (the
+/// tab-close flush) only POSTs.
+pub fn handle_save_draft(
+  req: Request,
+  config: ServerConfig,
+  code: String,
+) -> Response {
+  use json_body <- wisp.require_json(req)
+  case decode.run(json_body, draft_payload_decoder()) {
+    Error(_) -> wisp.json_response(json_ok(False), 400)
+    Ok(payload) -> {
+      let applicant_email = string.trim(payload.email)
+      case email.is_valid(applicant_email) {
+        False -> wisp.json_response(json_ok(False), 400)
+        True ->
+          case lookup_accepting_vacancy(config.db, code) {
+            Error(resp) -> resp
+            Ok(#(_vacancy, vacancy_uuid)) -> {
+              let filtered = filter_fields(payload.fields)
+              case filtered {
+                // An empty/fully-filtered map must NOT wipe a real draft —
+                // upserting '{}' would erase the applicant's saved answers
+                // (an unauthenticated destructive write needing only the
+                // victim's email + the public vacancy code).
+                [] -> wisp.json_response(json_ok(True), 200)
+                _ ->
+                  case
+                    sql.upsert_application_draft(
+                      config.db,
+                      vacancy_uuid,
+                      applicant_email,
+                      fields_to_json(filtered),
+                      wisp.random_string(48),
+                    )
+                  {
+                    Ok(_) -> {
+                      // First save after a resume stamps resumed_at (honest
+                      // metric: re-engagement, never the prefetch-able GET).
+                      // Fail-open.
+                      case payload.resume_token {
+                        "" -> Nil
+                        token -> {
+                          let _ =
+                            sql.mark_draft_resumed_by_token(
+                              config.db,
+                              token,
+                              vacancy_uuid,
+                            )
+                          Nil
+                        }
+                      }
+                      wisp.json_response(json_ok(True), 200)
+                    }
+                    Error(err) -> {
+                      wisp.log_error(
+                        "draft upsert failed: " <> string.inspect(err),
+                      )
+                      wisp.internal_server_error()
+                    }
+                  }
+              }
+            }
+          }
+      }
+    }
+  }
+}

+/// POST /apply/:code/continue-link — save current state, then email the
+/// tokenized continue link (with the erasure link riding along). The
+/// response carries NO send/cooldown state (ok:true on every claim outcome;
+/// ok:false only when the Resend call itself failed) — a flag would oracle
+/// the victim's draft/submission state to an empty-fields poller (N14,
+/// Perkins r2). Claim DB failures are 500.
+pub fn handle_continue_link(
+  req: Request,
+  config: ServerConfig,
+  code: String,
+) -> Response {
+  use json_body <- wisp.require_json(req)
+  case decode.run(json_body, draft_payload_decoder()) {
+    Error(_) -> wisp.json_response(json_ok(False), 400)
+    Ok(payload) -> {
+      let applicant_email = string.trim(payload.email)
+      case email.is_valid(applicant_email) {
+        False -> wisp.json_response(json_ok(False), 400)
+        True ->
+          case lookup_accepting_vacancy(config.db, code) {
+            Error(resp) -> resp
+            Ok(#(vacancy, vacancy_uuid)) -> {
+              // Save-then-send: the link must never restore less than what
+              // was on screen when the applicant asked for it. An empty
+              // field map (defensive client) must NOT wipe a real draft, so
+              // the upsert only runs when fields actually arrived — and a
+              // FAILED upsert aborts before the claim (a stale-state link
+              // is worse than no link).
+              let filtered = filter_fields(payload.fields)
+              let upserted = case filtered {
+                [] -> Ok(Nil)
+                _ ->
+                  sql.upsert_application_draft(
+                    config.db,
+                    vacancy_uuid,
+                    applicant_email,
+                    fields_to_json(filtered),
+                    wisp.random_string(48),
+                  )
+                  |> result.replace(Nil)
+                  |> result.map_error(fn(err) {
+                    wisp.log_error(
+                      "draft upsert (continue-link) failed: "
+                      <> string.inspect(err),
+                    )
+                    Nil
+                  })
+              }
+              case upserted {
+                Error(_) -> wisp.json_response(json_ok(False), 200)
+                Ok(_) -> {
+                  // Same re-engagement stamp as the plain save path.
+                  case payload.resume_token {
+                    "" -> Nil
+                    token -> {
+                      let _ =
+                        sql.mark_draft_resumed_by_token(
+                          config.db,
+                          token,
+                          vacancy_uuid,
+                        )
+                      Nil
+                    }
+                  }
+                  case
+                    sql.claim_continue_link_send(
+                      config.db,
+                      vacancy_uuid,
+                      applicant_email,
+                      wisp.random_string(48),
+                    )
+                  {
+                    Error(err) -> {
+                      wisp.log_error(
+                        "continue-link claim failed: " <> string.inspect(err),
+                      )
+                      wisp.internal_server_error()
+                    }
+                    // Cooldown, already submitted, or the empty-fields call
+                    // found no draft: identical response either way (N14 —
+                    // a `sent` flag would oracle the victim's
+                    // draft/submission state to an empty-fields poller).
+                    Ok(pog.Returned(_, [])) ->
+                      wisp.json_response(json_ok(True), 200)
+                    Ok(pog.Returned(_, [row, ..])) -> {
+                      // B1 (Perkins r1): emailed capability URLs come
+                      // from vacancy.origin_domain — NEVER the request
+                      // origin. canonical_origin trusts any *.run.app
+                      // X-Forwarded-Host; on this public endpoint that
+                      // would let an attacker ship the victim's genuine
+                      // Resend email with THEIR token on HIS domain. (The
+                      // confirmation email's /erase/ link precedent.)
+                      let base =
+                        email_client.build_entity_url(
+                          origin_domain: vacancy.origin_domain,
+                          path: "/apply/" <> vacancy.short_code,
+                        )
+                      case
+                        email_client.send_draft_continue_link(
+                          config.resend_api_key,
+                          applicant_email,
+                          vacancy.property_name,
+                          base <> "/resume/" <> row.continue_token,
+                          base <> "/draft-erasure/" <> row.continue_token,
+                          vacancy.closes_at,
+                          vacancy.id,
+                        )
+                      {
+                        Ok(_) -> wisp.json_response(json_ok(True), 200)
+                        Error(err) -> {
+                          wisp.log_error("continue-link send failed: " <> err)
+                          // The claim stamped the cooldown before the send;
+                          // clear it or the applicant's retry would be
+                          // cooldown-skipped while the UI says "try again".
+                          // Fail-open (the stamp self-heals in 10 minutes).
+                          let _ =
+                            sql.clear_last_link_sent(
+                              config.db,
+                              vacancy_uuid,
+                              applicant_email,
+                            )
+                          wisp.json_response(json_ok(False), 200)
+                        }
+                      }
+                    }
+                  }
+                }
+              }
+            }
+          }
+      }
+    }
+  }
+}
